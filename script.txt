local SavedEffects = {}

local function SetFPS(value)

	FPSBoost = value

	if value then

		for _,obj in ipairs(Lighting:GetChildren()) do

			if obj:IsA("BloomEffect")
			or obj:IsA("BlurEffect")
			or obj:IsA("ColorCorrectionEffect")
			or obj:IsA("SunRaysEffect")
			or obj:IsA("DepthOfFieldEffect") then

				SavedEffects[obj] = obj.Enabled
				obj.Enabled = false
			end
		end

		pcall(function()

			local Terrain =
				workspace:FindFirstChildOfClass("Terrain")

			if Terrain then
				Terrain.Decoration = false
				Terrain.WaterWaveSize = 0
				Terrain.WaterWaveSpeed = 0
				Terrain.WaterReflectance = 0
			end
		end)

		pcall(function()
			settings().Rendering.QualityLevel =
				Enum.QualityLevel.Level01
		end)

		FPSButton.Text = "⚡ FPS BOOST : ON"
		FPSButton.BackgroundColor3 =
			Color3.fromRGB(45,105,60)

	else

		for obj,state in pairs(SavedEffects) do

			if obj and obj.Parent then
				obj.Enabled = state
			end
		end

		table.clear(SavedEffects)

		FPSButton.Text = "⚡ FPS BOOST : OFF"
		FPSButton.BackgroundColor3 = PANEL2
	end
end

FPSButton.MouseButton1Click:Connect(function()
	SetFPS(not FPSBoost)
end)

local dragging = false

Header.InputBegan:Connect(function(input)

	if input.UserInputType ==
		Enum.UserInputType.MouseButton1
	or input.UserInputType ==
		Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = Main.Position
	end

	if input.UserInputType ==
		Enum.UserInputType.MouseButton1
	or input.UserInputType ==
		Enum.UserInputType.Touch then

		dragging = false
	end
end)

UIS.InputChanged:Connect(function(input)

	if not dragging then return end

	if input.UserInputType ==
		Enum.UserInputType.MouseMovement
	or input.UserInputType ==
		Enum.UserInputType.Touch then

		local delta =
			input.Position - dragStart

		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)

local function GetZone(index)

	if not Zones then
		return nil
	end

	local target = Vector3.new(
		L[index][1],
		L[index][2],
		L[index][3]
	)

	local nearest = nil
	local shortest = math.huge

	for _,zone in ipairs(Zones:GetChildren()) do

		local core =
			zone:FindFirstChild("LightCore")

		if core then

			local pos

			if zone:IsA("BasePart") then

				pos = zone.Position

			elseif zone:IsA("Model") then

				pos = zone:GetPivot().Position

			elseif core:IsA("BasePart") then

				pos = core.Position
			end

			if pos then

				local distance =
					(pos-target).Magnitude

				if distance < shortest then

					shortest = distance
					nearest = zone
				end
			end
		end
	end

	return nearest
end

local function IsGreen(core)

	if not core then
		return false
	end

	if not core:IsA("BasePart") then
		return false
	end

	local c = core.Color

	return c.R < .1
		and c.G > .9
		and c.B < .1
end

local function TeleportZone(index)

	local char = Player.Character
	if not char then return end

	local root =
		char:FindFirstChild("HumanoidRootPart")

	if not root then return end

	root.CFrame = CFrame.new(
		L[index][4],
		L[index][5],
		L[index][6]
	)

	root.AssemblyLinearVelocity = Vector3.zero
	root.AssemblyAngularVelocity = Vector3.zero
end

--==================================================
-- FIND NON-GREEN ZONE
--==================================================

local function FindFarmZone()

	if not Zones then
		return nil
	end

	for i = 1,9 do

		if FarmZone[i] then

			local zone = GetZone(i)

			local core =
				zone and
				zone:FindFirstChild("LightCore")

			if core and not IsGreen(core) then
				return i
			end
		end
	end

	return nil
end

-- AUTO FARM
--==================================================

task.spawn(function()

	while task.wait(.25) do

		if AutoFarm then

			local zoneIndex =
				FindFarmZone()

			if zoneIndex then

				local zone =
					GetZone(zoneIndex)

				local core =
					zone and
					zone:FindFirstChild("LightCore")

				if core then

					TeleportZone(zoneIndex)

					QuestStatus.Text =
						"Đang farm Zone "..zoneIndex

					QuestStatus.TextColor3 =
						LIGHT_PINK

					while AutoFarm
					and FarmZone[zoneIndex]
					and core
					and not IsGreen(core) do

						local char =
							Player.Character

						local root =
							char and
							char:FindFirstChild(
								"HumanoidRootPart"
							)

						if root then

							root.CFrame =
								CFrame.new(
									L[zoneIndex][4],
									L[zoneIndex][5],
									L[zoneIndex][6]
								)

							root.AssemblyLinearVelocity =
								Vector3.zero

							root.AssemblyAngularVelocity =
								Vector3.zero
						end

						task.wait(.15)
					end
				end

			else

				-- SAFE POSITION = ZONE 3
				local char =
					Player.Character

				local root =
					char and
					char:FindFirstChild(
						"HumanoidRootPart"
					)

				if root then

					local safe =
						Vector3.new(
							L[3][4],
							L[3][5],
							L[3][6]
						)

					if (root.Position-safe).Magnitude > 5 then

						root.CFrame =
							CFrame.new(safe)

						root.AssemblyLinearVelocity =
							Vector3.zero

						root.AssemblyAngularVelocity =
							Vector3.zero
					end
				end

				QuestStatus.Text =
					"Tất cả Zone xanh • Z3"

				QuestStatus.TextColor3 =
					Color3.fromRGB(80,255,120)
			end
		end
	end
end)

-- QUEST DETECTION
--==================================================

task.spawn(function()

	while task.wait(.3) do

		pcall(function()

			local gui =
				Player:FindFirstChild("PlayerGui")

			local gameUI =
				gui and
				gui:FindFirstChild("GameUI")

			local quests =
				gameUI and
				gameUI:FindFirstChild("QuestsFrame")

			local detected = "None"

			if quests then

				for _,child in ipairs(
					quests:GetChildren()
				) do

					local text = ""

					for _,desc in ipairs(
						child:GetDescendants()
					) do

						if desc:IsA("TextLabel") then

							text =
								text.." "
								..string.lower(
									desc.Text
								)
						end
					end

					if string.find(
						text,
						"eat something"
					) then

						detected = "Eat"
						break

					elseif string.find(
						text,
						"drink water"
					) then

						detected = "Drink"
						break

					elseif string.find(
						text,
						"rest"
					) then

						detected = "Rest"
						break
					end
				end
			end

			_G.CurrentQuest = detected

			if detected == "Eat" then

				QuestStatus.Text = "Quest: EAT"
				QuestStatus.TextColor3 =
					Color3.fromRGB(80,255,80)

				QuestPageStatus.Text =
					"Current Quest: Eat"

				QuestPageStatus.TextColor3 =
					Color3.fromRGB(80,255,80)

			elseif detected == "Drink" then

				QuestStatus.Text = "Quest: DRINK"
				QuestStatus.TextColor3 =
					Color3.fromRGB(100,180,255)

				QuestPageStatus.Text =
					"Current Quest: Drink"

				QuestPageStatus.TextColor3 =
					Color3.fromRGB(100,180,255)

			elseif detected == "Rest" then

				QuestStatus.Text = "Quest: REST"
				QuestStatus.TextColor3 =
					Color3.fromRGB(255,180,100)

				QuestPageStatus.Text =
					"Current Quest: Rest"

				QuestPageStatus.TextColor3 =
					Color3.fromRGB(255,180,100)

			else

				QuestPageStatus.Text =
					"Current Quest: Waiting..."

			end
		end)
	end
end)

task.spawn(function()

	local remote

	pcall(function()
		remote =
			RS.Remotes.Character.Eat
	end)

	while task.wait(.4) do

		if _G.AutoEatRunning
		and _G.CurrentQuest == "Eat"
		and remote then

			pcall(function()

				local env =
					workspace:FindFirstChild(
						"MapEnvironment"
					)

				if not env then return end

				local char =
					Player.Character

				local root =
					char and
					char:FindFirstChild(
						"HumanoidRootPart"
					)

				if not root then return end

				local closest = nil
				local shortest = math.huge

				local folders = {

					{
						env:FindFirstChild("RedWood"),
						"Trees",
						"RedWoodTree"
					},

					{
						env:FindFirstChild("Pines"),
						"Trees",
						"PineTree"
					},

					{
						env:FindFirstChild("Desert"),
						"Tress",
						"DesertTree"
					},

					{
						env:FindFirstChild("Lobby"),
						"Trees",
						"RedWoodTree"
					}
				}

				for _,data in ipairs(folders) do

					local folder =
						data[1] and
						data[1]:FindFirstChild(data[2])

					if folder then

						for _,tree in ipairs(
							folder:GetChildren()
						) do

							if tree.Name == data[3] then

								local part

								if tree:IsA("BasePart") then
									part = tree
								else
									part =
										tree:FindFirstChildWhichIsA(
											"BasePart"
										)
								end

								if part then

									local d =
										(root.Position-
										part.Position).Magnitude

									if d < shortest then

										shortest = d
										closest = tree
									end
								end
							end
						end
					end
				end

				if closest then
					remote:FireServer(closest)
				end
			end)
		end
	end
end)

--==================================================
-- AUTO REST
--==================================================

task.spawn(function()

	local resting = false

	while task.wait(.3) do

		local shouldRest =
			_G.AutoRestRunning
			and _G.CurrentQuest == "Rest"

		pcall(function()

			local folder =
				RS:FindFirstChild("Remotes")

			local character =
				folder and
				folder:FindFirstChild("Character")

			local remote =
				character and
				character:FindFirstChild("Rest")

			if not remote then return end

			if shouldRest and not resting then

				remote:InvokeServer(true)
				resting = true

			elseif not shouldRest and resting then

				remote:InvokeServer(false)
				resting = false
			end
		end)
	end
end)
